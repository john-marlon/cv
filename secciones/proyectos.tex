\section{Proyectos destacados}

\begin{cvproject}{Sistema de gestión del soporte \mbox{post-venta} (SOPOS)}{Tesis de Ingeniería Informática}{Desoft, División Matanzas · Desarrollo propio}
\begin{itemize}
  \item Autor y desarrollador principal de una solución empresarial sobre Odoo 18, organizada en una suite de diez módulos personalizados; análisis del proceso, definición de requisitos, diseño del dominio y de la base de datos, e implementación con orientación de un tutor.
  \item Desarrollo de gestión de tickets y órdenes de servicio, asignación de especialistas, ciclo de vida de incidencias, escalamientos, portal de cliente y dashboard con OWL y Chart.js.
  \item Implementación de controles de acceso con grupos, ACL y reglas de registro, y del reporte PDF de escalamiento con QWeb; validación con pruebas de aceptación, compatibilidad entre navegadores y usabilidad. \textbf{Tecnologías:} Odoo 18, Python, PostgreSQL, JavaScript, HTML, CSS, XML, OWL, Chart.js, QWeb, Docker, Git, Scrum, UML y Visual Paradigm.
\end{itemize}
\end{cvproject}

\begin{cvproject}{PROVEA — Plataforma de actores económicos y oportunidades B2B}{Octubre de 2026 – Actualidad}{Desoft, División Matanzas · Desarrollo en equipo}
\begin{itemize}
  \item Plataforma empresarial sobre Odoo 18 para gestionar empresas cubanas, sus productos y servicios, y oportunidades comerciales B2B; suite modular de más de 30 módulos con clasificadores \mbox{CNAE/NAE/CPCU/OSDE/OACE}, formas organizativas, empresas, productos, CRM de leads, dashboard y API pública documentada con \mbox{OpenAPI/Swagger}.
  \item Contribución principal: geolocalización de empresas desde la dirección, con geocodificación en backend, mapa \mbox{OpenStreetMap} embebido, enlace a mapa externo y controlador de formulario Odoo 18 registrado mediante \mbox{js\_class}, sin parchear clases core.
  \item El botón actualiza las coordenadas como cambios pendientes y las persiste solo al guardar; resetea las coordenadas al cambiar la dirección y valida latitud/longitud. \textbf{Stack:} Odoo 18, Python, PostgreSQL, JavaScript/OWL, XML, \mbox{OpenStreetMap}, Google Maps, Docker y \mbox{Git/GitHub}.
\end{itemize}
\end{cvproject}
